MongoDB Aggregation Pipeline (Student 1 Analytics):

db.vehicle_maintenance.aggregate([
  { $unwind: "$maintenanceLogs" },
  { $match: { "maintenanceLogs.serviceCost": { $gt: 0 } } },
  { $group: {
      _id: {
        stationAddress: "$station.streetAddress",
        vin:            "$vin",
        modelType:      "$modelType"
      },
      totalLogs: { $sum: 1 },
      totalCost: { $sum: "$maintenanceLogs.serviceCost" },
      avgCost:   { $avg: "$maintenanceLogs.serviceCost" }
  }},
  { $sort: { totalCost: -1 } }
])
